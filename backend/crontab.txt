# Crontab bot absensi & jadwal WFO — semua jam ditulis dalam WIB.
#
# CRON_TZ mengunci jadwal ke Asia/Jakarta, jadi file ini berperilaku sama di mesin
# ber-TZ UTC maupun WIB. Tanpa baris itu, cron memakai zona waktu lokal host dan
# jamnya ikut bergeser. (CRON_TZ didukung cron bawaan Debian/Ubuntu dan cronie;
# kalau host memakai cron lain, pastikan dulu variabel ini dikenali.)
#
# Install: crontab backend/crontab.txt
# Cek hasilnya: crontab -l
# Sesuaikan PROJECT bila path deploy-nya berbeda.

CRON_TZ=Asia/Jakarta
TZ=Asia/Jakarta

PROJECT=/home/reyhanr/wknd/projects/wknd-tele-bot-employee-schedule/backend
LOG=$PROJECT/logs

# Sync jadwal dari Supabase — 20:00 WIB, setiap hari
0 20 * * * cd $PROJECT && npx tsx src/cron/sync-schedules.ts >> $LOG/sync-schedules.log 2>&1

# Cek validitas token Google — 08:00 WIB, Senin-Jumat
0 8 * * 1-5 cd $PROJECT && npx tsx src/cron/check-tokens.ts >> $LOG/check-tokens.log 2>&1

# Reminder check-in — 09:30 dan 09:50 WIB, Senin-Jumat
30 9 * * 1-5 cd $PROJECT && npx tsx src/cron/reminder.ts checkin >> $LOG/reminder.log 2>&1
50 9 * * 1-5 cd $PROJECT && npx tsx src/cron/reminder.ts checkin >> $LOG/reminder.log 2>&1

# Reminder check-out — 18:00, 21:00, dan 23:00 WIB, Senin-Jumat
0 18,21,23 * * 1-5 cd $PROJECT && npx tsx src/cron/reminder.ts checkout >> $LOG/reminder.log 2>&1

# Reminder WFO besok — 21:00 WIB, Senin-Kamis
0 21 * * 1-4 cd $PROJECT && npx tsx src/cron/reminder-wfo.ts tomorrow >> $LOG/reminder-wfo.log 2>&1

# Reminder jadwal minggu depan — 21:00 WIB, Jumat
0 21 * * 5 cd $PROJECT && npx tsx src/cron/reminder-wfo.ts weekly >> $LOG/reminder-wfo.log 2>&1
